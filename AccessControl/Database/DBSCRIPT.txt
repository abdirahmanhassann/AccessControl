CREATE OR ALTER PROCEDURE [dbo].[AccessRequestApprovalsGet]
    @Id INT = NULL,
    @AccessRequestId INT = NULL,
    @ApproverUserId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        ara.[Id],
        ara.[AccessRequestId],
        ara.[ApproverUserId],
        u.[FirstName] as [ApproverFirstName],
        u.[LastName] as [ApproverLastName],
        u.[Email] as [ApproverEmail],
        w.[FirstName] as [WorkerFirstName],
        w.[LastName] as [WorkerLastName],
        w.[PhoneNumber] as [WorkerPhoneNumber],
        ara.[Status],
        ara.[Comment],
        ara.[ReviewedAt],
        ar.[CreatedAt] as [AccessrequestCreatedAt]
    FROM [dbo].[AccessRequestApprovals] ara
    INNER JOIN [dbo].[Users] u
        ON u.[Id] = ara.[ApproverUserId]
    INNER JOIN [dbo].[AccessRequests] ar
        ON ar.id = ara.AccessRequestId
    INNER JOIN [dbo].[Workers] w 
        ON w.id = ar.WorkerId
    WHERE (@Id IS NULL OR ara.[Id] = @Id)
      AND (@AccessRequestId IS NULL OR ara.[AccessRequestId] = @AccessRequestId)
      AND (@ApproverUserId IS NULL OR ara.[ApproverUserId] = @ApproverUserId)
    ORDER BY ar.[CreatedAt] DESC;
END;

CREATE OR ALTER PROCEDURE [dbo].[AccessRequestsGet]
    @Id INT = NULL,
    @WorkerId INT = NULL,
    @RoomId INT = NULL,
    @Status NVARCHAR(30) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        ar.[Id],

        ar.[WorkerId],
        w.[FirstName] AS [WorkerFirstName],
        w.[LastName] AS [WorkerLastName],
        w.[PhoneNumber] AS [WorkerPhoneNumber],
        w.[CompanyName],

        ar.[RoomId],
        r.[RoomNumber],
        r.[Name] AS [RoomName],

        wa.[Id] AS [WorkAreaId],

        ar.[Status],
        ar.[Reason],
        ar.[WorkType],
        ar.[Description],
        ar.[RequestedAt],
        ar.[ApprovedAt],
        ar.[RejectedAt],
        ar.[ClockedInAt],
        ar.[ExpectedClockOutAt],
        ar.[ClockedOutAt],
        ar.[CompletedAt]

    FROM [dbo].[AccessRequests] ar

    INNER JOIN [dbo].[Workers] w
        ON w.[Id] = ar.[WorkerId]

    INNER JOIN [dbo].[Rooms] r
        ON r.[Id] = ar.[RoomId]

    INNER JOIN [dbo].[WorkAreas] wa
        ON wa.[Id] = r.[WorkAreaId]

      AND (@WorkerId IS NULL OR ar.[WorkerId] = @WorkerId)
      AND (@RoomId IS NULL OR ar.[RoomId] = @RoomId)
      AND (@Status IS NULL OR ar.[Status] = @Status)

CREATE OR ALTER PROCEDURE [dbo].[WorkAreasGet]
    @Id INT = NULL,
    @SiteId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        wa.[Id],
        wa.[SiteId],
        s.[Name] AS [SiteName],
        wa.[Name],
        wa.[Description],
        wa.[IsActive],
        wa.[CreatedAt]
    FROM [dbo].[WorkAreas] wa
    INNER JOIN [dbo].[Sites] s
        ON s.[Id] = wa.[SiteId]
    WHERE (@Id IS NULL OR wa.[Id] = @Id)
      AND (@SiteId IS NULL OR @SiteId = 0 OR wa.[SiteId] = @SiteId)
    ORDER BY s.[Name], wa.[Name];
END;

CREATE OR ALTER PROCEDURE [dbo].[ManagersForRoomGet]
    @RoomId INT,
    @DepartmentId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF @DepartmentId IS NOT NULL AND @DepartmentId > 0
    BEGIN
        SELECT DISTINCT
            u.[Id],
            u.[Id] AS [UserId],
            u.[FirstName],
            u.[LastName],
            u.[Email],
            u.[PhoneNumber],
            u.[Role],
            u.[IsActive],
            d.[Id] AS [DepartmentId],
            d.[Name] AS [DepartmentName],
            d.[SiteId],
            s.[Name] AS [SiteName]
        FROM [dbo].[DepartmentManagerMapping] dmm
        INNER JOIN [dbo].[Departments] d ON d.[Id] = dmm.[DepartmentId] AND d.[IsActive] = 1
        INNER JOIN [dbo].[Users] u ON u.[Id] = dmm.[UserId]
        LEFT JOIN [dbo].[Sites] s ON s.[Id] = d.[SiteId]
        WHERE dmm.[DepartmentId] = @DepartmentId
        ORDER BY u.[LastName], u.[FirstName];
        RETURN;
    END

    IF EXISTS (SELECT 1 FROM [dbo].[RoomManagers] WHERE [RoomId] = @RoomId)
    BEGIN
        SELECT DISTINCT
            u.[Id],
            u.[Id] AS [UserId],
            u.[FirstName],
            u.[LastName],
            u.[Email],
            u.[PhoneNumber],
            u.[Role],
            u.[IsActive],
            CAST(NULL AS INT) AS [DepartmentId],
            CAST(N'Location' AS NVARCHAR(200)) AS [DepartmentName],
            wa.[SiteId],
            s.[Name] AS [SiteName]
        FROM [dbo].[RoomManagers] rm
        INNER JOIN [dbo].[Users] u ON u.[Id] = rm.[UserId] AND u.[IsActive] = 1
        INNER JOIN [dbo].[Rooms] r ON r.[Id] = rm.[RoomId]
        INNER JOIN [dbo].[WorkAreas] wa ON wa.[Id] = r.[WorkAreaId]
        LEFT JOIN [dbo].[Sites] s ON s.[Id] = wa.[SiteId]
        WHERE rm.[RoomId] = @RoomId
        ORDER BY u.[LastName], u.[FirstName];
        RETURN;
    END

    SELECT DISTINCT
        u.[Id],
        u.[Id] AS [UserId],
        u.[FirstName],
        u.[LastName],
        u.[Email],
        u.[PhoneNumber],
        u.[Role],
        u.[IsActive],
        d.[Id] AS [DepartmentId],
        d.[Name] AS [DepartmentName],
        d.[SiteId],
        s.[Name] AS [SiteName]
    FROM [dbo].[DepartmentManagerMapping] dmm
    INNER JOIN [dbo].[Departments] d ON d.[Id] = dmm.[DepartmentId] AND d.[IsActive] = 1
    INNER JOIN [dbo].[Users] u ON u.[Id] = dmm.[UserId]
    INNER JOIN [dbo].[DepartmentRoomMapping] drm ON drm.[DepartmentId] = d.[Id]
    LEFT JOIN [dbo].[Sites] s ON s.[Id] = d.[SiteId]
    WHERE drm.[RoomId] = @RoomId
    ORDER BY u.[LastName], u.[FirstName];
END

CREATE OR ALTER PROCEDURE [dbo].[ManagerCanApprove]
    @ApproverUserId INT,
    @RoomId INT = NULL,
    @AccessRequestId INT = NULL,
    @DepartmentId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @ResolvedRoomId INT = @RoomId;
    IF @ResolvedRoomId IS NULL AND @AccessRequestId IS NOT NULL
        SELECT @ResolvedRoomId = [RoomId] FROM [dbo].[AccessRequests] WHERE [Id] = @AccessRequestId;

    SELECT CAST(CASE WHEN EXISTS (
        SELECT 1
        FROM [dbo].[RoomManagers] rm
        WHERE rm.[UserId] = @ApproverUserId
          AND rm.[RoomId] = @ResolvedRoomId
    ) OR EXISTS (
        SELECT 1
        FROM [dbo].[DepartmentManagerMapping] dmm
        INNER JOIN [dbo].[Departments] d ON d.[Id] = dmm.[DepartmentId] AND d.[IsActive] = 1
        INNER JOIN [dbo].[DepartmentRoomMapping] drm ON drm.[DepartmentId] = d.[Id]
        WHERE dmm.[UserId] = @ApproverUserId
          AND drm.[RoomId] = @ResolvedRoomId
    ) OR EXISTS (
        SELECT 1
        FROM [dbo].[DepartmentManagerMapping] dmm
        INNER JOIN [dbo].[Departments] d ON d.[Id] = dmm.[DepartmentId] AND d.[IsActive] = 1
        WHERE dmm.[UserId] = @ApproverUserId
          AND @DepartmentId IS NOT NULL
          AND @DepartmentId > 0
          AND dmm.[DepartmentId] = @DepartmentId
    ) OR EXISTS (
        SELECT 1
        FROM [dbo].[WorkAreaManagers] wam
        INNER JOIN [dbo].[Rooms] r ON r.[WorkAreaId] = wam.[WorkAreaId]
        WHERE wam.[UserId] = @ApproverUserId
          AND r.[Id] = @ResolvedRoomId
    ) THEN 1 ELSE 0 END AS INT) AS [Allowed];
END
